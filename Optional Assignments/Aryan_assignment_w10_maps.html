import geopandas as gpd
import pandas as pd
import random
import altair as alt
import json

us_map = gpd.read_file("ne_110m_admin_1_states_provinces/ne_110m_admin_1_states_provinces.shp")
us_map = us_map[(us_map['admin'] == "United States of America") & (~us_map['name'].isin(['Alaska', 'Hawaii']))]

case_categories = {
    "None": (0, 0),
    "1 to 100": (1, 100),
    "101 to 1,000": (101, 1000),
    "1,001 to 5,000": (1001, 5000),
    "5,001 to 10,000": (5001, 10000),
    "10,001 or more": (10001, 50000)
}
case_data = {state: random.randint(*case_categories[random.choice(list(case_categories.keys()))])
             for state in us_map['name']}

us_map['traffic'] = us_map['name'].map(case_data)

us_map_json = json.loads(us_map.to_json())

chart = alt.Chart(alt.Data(values=us_map_json['features'])).mark_geoshape(
    stroke='black'
).encode(
    color=alt.Color('properties.traffic:Q', scale=alt.Scale(scheme='blues'), title='Traffic'),
    
    tooltip=['properties.name:N', 'properties.traffic:Q']
).project(
    type='albersUsa'
).properties(
    width=800,
    height=500,
    title=f'Traffic Distribution by State (Total: {sum(case_data.values()):,})'
)

chart.display()

Discussion¶

  • The map visualizes traffic volume across U.S. states, with darker blue shades indicating higher traffic.

  • States like California, Georgia, New York, Illinois, and Missouri show the highest traffic, as seen by the darkest blue regions.

  • Most other states have moderate to low traffic, represented by lighter blue shades.

  • The total traffic across all states is 428,264, as noted in the title.

  • This distribution suggests that high-population or high-urbanization states tend to experience more traffic congestion.
